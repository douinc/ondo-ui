## 설치

```bash
bunx shadcn@latest add @ondo-ui/textarea
```

## 사용법

```tsx
import { Textarea } from "@/components/ui/textarea"
```

```tsx
<Textarea />
```

## 예시

### 크기

`xs`, `sm`, `default`, `lg`, `xl`, `2xl`을 지원합니다.

가로 패딩·글자 크기·모서리 반경이 모든 단계에서 [`Input`](/docs/components/input)과 일치하므로, 같은 크기의 textarea와 input을 폼에 나란히 두면 좌우가 맞습니다. 여러 줄 입력이라 세로 패딩과 최소 높이만 다릅니다.

<ComponentPreview name="textarea-sizes">

```tsx
import { Textarea } from "@/components/ui/textarea"

const sizes = ["xs", "sm", "default", "lg", "xl", "2xl"] as const

export default function TextareaSizes() {
  return (
    <div className="flex w-full max-w-sm flex-col gap-3">
      {sizes.map((size) => (
        <Textarea key={size} size={size} placeholder={`Size ${size}`} />
      ))}
    </div>
  )
}
```

</ComponentPreview>

### 필드

`Field`, `FieldLabel`, `FieldDescription`을 사용하여 레이블과 설명이 있는 텍스트 영역을 만듭니다.

<ComponentPreview name="textarea-field">

```tsx
import {
  Field,
  FieldDescription,
  FieldLabel,
} from "@/components/ui/field"
import { Textarea } from "@/components/ui/textarea"

export function TextareaField() {
  return (
    <Field>
      <FieldLabel htmlFor="textarea-message">Message</FieldLabel>
      <FieldDescription>Enter your message below.</FieldDescription>
      <Textarea id="textarea-message" placeholder="Type your message here." />
    </Field>
  )
}
```

</ComponentPreview>

### 비활성화됨

`disabled` prop을 사용하여 텍스트 영역을 비활성화합니다. 비활성화 상태의 스타일을 지정하려면 `Field` 컴포넌트에 `data-disabled` 속성을 추가합니다.

<ComponentPreview name="textarea-disabled">

```tsx
import { Field, FieldLabel } from "@/components/ui/field"
import { Textarea } from "@/components/ui/textarea"

export function TextareaDisabled() {
  return (
    <Field data-disabled>
      <FieldLabel htmlFor="textarea-disabled">Message</FieldLabel>
      <Textarea
        id="textarea-disabled"
        placeholder="Type your message here."
        disabled
      />
    </Field>
  )
}
```

</ComponentPreview>

### 유효하지 않음

`aria-invalid` prop을 사용하여 텍스트 영역을 유효하지 않은 것으로 표시합니다. 유효하지 않은 상태의 스타일을 지정하려면 `Field` 컴포넌트에 `data-invalid` 속성을 추가합니다.

<ComponentPreview name="textarea-invalid">

```tsx
import {
  Field,
  FieldDescription,
  FieldLabel,
} from "@/components/ui/field"
import { Textarea } from "@/components/ui/textarea"

export function TextareaInvalid() {
  return (
    <Field data-invalid>
      <FieldLabel htmlFor="textarea-invalid">Message</FieldLabel>
      <Textarea
        id="textarea-invalid"
        placeholder="Type your message here."
        aria-invalid
      />
      <FieldDescription>Please enter a valid message.</FieldDescription>
    </Field>
  )
}
```

</ComponentPreview>

### 버튼

`Button`과 페어링하여 제출 버튼이 있는 텍스트 영역을 만듭니다.

<ComponentPreview name="textarea-button">

```tsx
import { Button } from "@/components/ui/button"
import { Textarea } from "@/components/ui/textarea"

export function TextareaButton() {
  return (
    <div className="grid w-full gap-2">
      <Textarea placeholder="Type your message here." />
      <Button>Send message</Button>
    </div>
  )
}
```

</ComponentPreview>

## RTL

shadcn/ui에서 RTL 지원을 활성화하려면 [RTL 구성 가이드](https://ui.shadcn.com/docs/rtl)를 참조하세요.

```tsx
"use client"

import * as React from "react"

import {
  useTranslation,
  type Translations,
} from "@/components/language-selector"
import {
  Field,
  FieldDescription,
  FieldLabel,
} from "@/components/ui/field"
import { Textarea } from "@/components/ui/textarea"

const translations: Translations = {
  en: {
    dir: "ltr",
    values: {
      label: "Feedback",
      placeholder: "Your feedback helps us improve...",
      description: "Share your thoughts about our service.",
    },
  },
  ar: {
    dir: "rtl",
    values: {
      label: "التعليقات",
      placeholder: "تعليقاتك تساعدنا على التحسين...",
      description: "شاركنا أفكارك حول خدمتنا.",
    },
  },
  he: {
    dir: "rtl",
    values: {
      label: "משוב",
      placeholder: "המשוב שלך עוזר לנו להשתפר...",
      description: "שתף את מחשבותיך על השירות שלנו.",
    },
  },
}

export function TextareaRtl() {
  const { dir, t } = useTranslation(translations, "ar")

  return (
    <Field className="w-full max-w-xs" dir={dir}>
      <FieldLabel htmlFor="feedback" dir={dir}>
        {t.label}
      </FieldLabel>
      <Textarea id="feedback" placeholder={t.placeholder} dir={dir} rows={4} />
      <FieldDescription dir={dir}>{t.description}</FieldDescription>
    </Field>
  )
}
```
