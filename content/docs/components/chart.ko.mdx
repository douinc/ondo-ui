```tsx
"use client"

import * as React from "react"
import { Bar, BarChart, CartesianGrid, XAxis } from "recharts"

import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"
import {
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,
  type ChartConfig,
} from "@/components/ui/chart"

const chartData = [
  { date: "2024-04-01", desktop: 222, mobile: 150 },
  { date: "2024-04-02", desktop: 97, mobile: 180 },
  { date: "2024-04-03", desktop: 167, mobile: 120 },
  { date: "2024-04-04", desktop: 242, mobile: 260 },
  { date: "2024-04-05", desktop: 373, mobile: 290 },
  { date: "2024-04-06", desktop: 301, mobile: 340 },
  { date: "2024-04-07", desktop: 245, mobile: 180 },
]

const chartConfig = {
  views: {
    label: "페이지 조회수",
  },
  desktop: {
    label: "데스크톱",
    color: "var(--chart-2)",
  },
  mobile: {
    label: "모바일",
    color: "var(--chart-1)",
  },
} satisfies ChartConfig

export default function ChartDemo() {
  const [activeChart, setActiveChart] =
    React.useState<keyof typeof chartConfig>("desktop")

  const total = React.useMemo(
    () => ({
      desktop: chartData.reduce((acc, curr) => acc + curr.desktop, 0),
      mobile: chartData.reduce((acc, curr) => acc + curr.mobile, 0),
    }),
    []
  )

  return (
    <Card className="w-full py-0 pb-4">
      <CardHeader className="flex flex-col items-stretch border-b p-0! sm:flex-row">
        <div className="flex flex-1 flex-col justify-center gap-1 px-6 pt-4 pb-3 sm:py-0!">
          <CardTitle>막대 차트 - 인터랙티브</CardTitle>
          <CardDescription>
            최근 3개월간 전체 방문자 수
          </CardDescription>
        </div>
        <div className="flex">
          {["desktop", "mobile"].map((key) => {
            const chart = key as keyof typeof chartConfig
            return (
              <button
                key={chart}
                data-active={activeChart === chart}
                className="relative z-30 flex flex-1 flex-col justify-center gap-1 border-t px-6 py-4 text-left even:border-l data-[active=true]:bg-muted/50 sm:border-t-0 sm:border-l sm:px-8 sm:py-6"
                onClick={() => setActiveChart(chart)}
              >
                <span className="text-xs text-muted-foreground">
                  {chartConfig[chart].label}
                </span>
                <span className="text-lg leading-none font-bold sm:text-3xl">
                  {total[key as keyof typeof total].toLocaleString()}
                </span>
              </button>
            )
          })}
        </div>
      </CardHeader>
      <CardContent className="px-2 sm:p-6">
        <ChartContainer
          config={chartConfig}
          className="aspect-auto h-[250px] w-full"
        >
          <BarChart
            accessibilityLayer
            data={chartData}
            margin={{ left: 12, right: 12 }}
          >
            <CartesianGrid vertical={false} />
            <XAxis
              dataKey="date"
              tickLine={false}
              axisLine={false}
              tickMargin={8}
              minTickGap={32}
              tickFormatter={(value) => {
                const date = new Date(value)
                return date.toLocaleDateString("en-US", {
                  month: "short",
                  day: "numeric",
                })
              }}
            />
            <ChartTooltip
              content={
                <ChartTooltipContent
                  className="w-[150px]"
                  nameKey="views"
                  labelFormatter={(value) => {
                    return new Date(value).toLocaleDateString("en-US", {
                      month: "short",
                      day: "numeric",
                      year: "numeric",
                    })
                  }}
                />
              }
            />
            <Bar dataKey={activeChart} fill={`var(--color-${activeChart})`} />
          </BarChart>
        </ChartContainer>
      </CardContent>
    </Card>
  )
}
```

차트는 기본 상태로도 보기 좋게 설계되었습니다. 다른 컴포넌트와 잘 어울리며, 프로젝트에 맞게 자유롭게 커스터마이즈할 수 있습니다.

## 컴포넌트

내부적으로 [Recharts](https://recharts.org/)를 사용합니다.

`chart` 컴포넌트는 조합을 염두에 두고 설계했습니다. **차트는 Recharts 컴포넌트로 직접 만들고, `ChartTooltip` 같은 커스텀 컴포넌트는 필요한 곳에만 가져다 씁니다**.

```tsx showLineNumbers
import { Bar, BarChart } from "recharts"

import { ChartContainer, ChartTooltipContent } from "@/components/ui/chart"

export function MyChart() {
  return (
    <ChartContainer>
      <BarChart data={data}>
        <Bar dataKey="value" />
        <ChartTooltip content={<ChartTooltipContent />} />
      </BarChart>
    </ChartContainer>
  )
}
```

Recharts를 감싸지 않습니다. 즉 특정 추상화에 갇히지 않습니다. Recharts 새 버전이 나오면 공식 업그레이드 경로를 그대로 따라가면 됩니다.

**컴포넌트는 여러분의 것입니다**.

## Recharts v3로 업데이트하기

기존 차트 코드를 Recharts v3로 업데이트한다면:

- CSS 변수에서 차트 토큰을 참조할 때 `hsl(var(--chart-1))` 대신 `var(--chart-1)`을 사용하세요.
- `ChartTooltip.defaultIndex`는 초기 툴팁 상태에만 사용하세요. 지속적으로 활성화되는 도형은 직접 차트 상태로 관리하세요.
- 부모 `<BarChart>`에 이미 `layout`이 정의되어 있다면 `<Bar>`에서는 제거하세요.
- `ResponsiveContainer`가 첫 렌더링에서 크기를 측정할 수 있도록 `ChartContainer`에 높이나 `min-h-*`, `aspect-*`를 유지하세요.

## 차트 설정

차트 설정은 차트의 라벨, 아이콘, 색상을 정의하는 곳입니다.

차트 데이터와는 의도적으로 분리되어 있습니다.

덕분에 설정과 색상 토큰을 여러 차트에서 공유할 수 있고, 데이터나 색상 토큰이 원격에 있거나 다른 형식일 때도 독립적으로 동작합니다.

```tsx showLineNumbers
import { IconDeviceDesktop } from "@tabler/icons-react"

import { type ChartConfig } from "@/components/ui/chart"

const chartConfig = {
  desktop: {
    label: "데스크톱",
    icon: IconDeviceDesktop,
    // A color like 'hsl(220, 98%, 61%)' or 'var(--color-name)'
    color: "#2563eb",
    // OR a theme object with 'light' and 'dark' keys
    theme: {
      light: "#2563eb",
      dark: "#dc2626",
```

## 툴팁

차트 툴팁은 라벨, 이름, 표식, 값으로 구성됩니다. 이들을 조합해 툴팁을 원하는 대로 구성할 수 있습니다.

<ComponentPreview name="chart-tooltip">

```tsx
"use client"

import { Bar, BarChart, XAxis } from "recharts"

import {
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,
  type ChartConfig,
} from "@/components/ui/chart"

const chartData = [
  { month: "January", desktop: 186, mobile: 80 },
  { month: "February", desktop: 305, mobile: 200 },
  { month: "March", desktop: 237, mobile: 120 },
]

const chartConfig = {
  desktop: { label: "데스크톱", color: "var(--chart-1)" },
  mobile: { label: "모바일", color: "var(--chart-2)" },
} satisfies ChartConfig

export default function ChartTooltipDemo() {
  return (
    <ChartContainer config={chartConfig} className="h-[200px] w-full">
      <BarChart accessibilityLayer data={chartData}>
        <XAxis dataKey="month" tickLine={false} axisLine={false} />
        <ChartTooltip content={<ChartTooltipContent indicator="line" />} />
        <Bar dataKey="desktop" fill="var(--color-desktop)" radius={4} />
        <Bar dataKey="mobile" fill="var(--color-mobile)" radius={4} />
      </BarChart>
    </ChartContainer>
  )
}
```

</ComponentPreview>

`hideLabel`, `hideIndicator` 프롭으로 각 요소를 켜고 끌 수 있으며, `indicator` 프롭으로 표식 스타일을 지정합니다.

툴팁의 라벨과 이름에 다른 키를 쓰려면 `labelKey`와 `nameKey`를 사용하세요.

Chart는 `<ChartTooltip>`과 `<ChartTooltipContent>` 컴포넌트를 함께 제공합니다. 이 두 컴포넌트로 차트에 커스텀 툴팁을 추가할 수 있습니다.

```tsx title="components/example-chart.tsx"
import { ChartTooltip, ChartTooltipContent } from "@/components/ui/chart"
```

```tsx title="components/example-chart.tsx"
<ChartTooltip content={<ChartTooltipContent />} />
```

### 프롭

다음 프롭으로 툴팁을 구성합니다.

| 프롭            | 타입                     | 설명                                          |
| :-------------- | :----------------------- | :------------------------------------------- |
| `labelKey`      | string                   | 라벨에 사용할 설정 또는 데이터 키입니다.        |
| `nameKey`       | string                   | 이름에 사용할 설정 또는 데이터 키입니다.        |
| `indicator`     | `dot` `line` `dashed`    | 툴팁 표식 스타일입니다.                        |
| `hideLabel`     | boolean                  | 라벨을 숨길지 여부입니다.                      |
| `hideIndicator` | boolean                  | 표식을 숨길지 여부입니다.                      |

### 색상

색상은 차트 설정에서 자동으로 참조됩니다.

### 커스텀

툴팁 라벨과 이름에 다른 키를 쓰려면 `labelKey`와 `nameKey` 프롭을 사용하세요.

```tsx showLineNumbers
const chartData = [
  { browser: "chrome", visitors: 187, fill: "var(--color-chrome)" },
  { browser: "safari", visitors: 200, fill: "var(--color-safari)" },
]

const chartConfig = {
  visitors: {
    label: "전체 방문자",
  },
  chrome: {
    label: "Chrome",
    color: "var(--chart-1)",
  },
  safari: {
    label: "Safari",
    color: "var(--chart-2)",
  },
} satisfies ChartConfig
```

```tsx title="components/example-chart.tsx"
<ChartTooltip
  content={<ChartTooltipContent labelKey="visitors" nameKey="browser" />}
/>
```

이렇게 하면 라벨에는 `Total Visitors`가, 툴팁 이름에는 `Chrome`과 `Safari`가 사용됩니다.
